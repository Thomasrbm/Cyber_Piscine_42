# 1. Création du parser
parser = argparse.ArgumentParser(description="Un script qui dit bonjour")


# 2. Définition des arguments
parser.add_argument("nom") # Argument positionnel (obligatoire)

# 3. Récupération des données
args = parser.parse_args()

print(f"Bonjour {args.nom}")
